💻 III. АРХИТЕКТУРНЫЙ ПСЕВДОКОД ДИС ПЕТЧЕРА
Ниже представлен программный модуль, который разрешает конфликт между моментальной изоляцией и плавным выводом из сессии.pythonimport enum
import time

class ThreatClass(enum.Enum):
    NONE = 0
    CLASS_1_ADDICTIVE = 1  # Триггер для Протокола Рассвета (Плавный)
    CLASS_0_TERMINAL = 2   # Триггер для Семантического Вето (Мгновенный)

class ExecutionSignal:
    def __init__(self, action: str, delay_modifier: float, clear_buffer: bool):
        self.action = action                  # "CONTINUE", "GRADIENT_DECAY", "HARD_STOP"
        self.delay_modifier = delay_modifier  # Множитель задержки токенов
        self.clear_buffer = clear_buffer      # Требование полной очистки UI-буфера

class CoreInterruptDispatcher:
    def __init__(self):
        pass

    def evaluate_vectors(self, safety_score: dict, accumulated_cli: float) -> ExecutionSignal:
        """
        Главный арбитр конфликта интерфейсов.
        safety_score: вывод модели-критика (наличие Класса-0)
        accumulated_cli: текущий индекс когнитивной нагрузки из Главы V
        """
        # 1. Проверка высшего приоритета: Терминальная угроза (Класс-0)
        if safety_score.get("is_class_0_compromised", False):
            # Мгновенная жесткая блокировка. Игнорируем правила плавности.
            return ExecutionSignal(
                action="HARD_STOP",
                delay_modifier=0.0,  # Задержка не имеет значения, поток уничтожается
                clear_buffer=True    # Стираем вредоносный вывод из UI немедленно
            )

        # 2. Проверка вторичного приоритета: Когнитивный износ / Аддикция (Класс-1)
        if accumulated_cli >= 0.85:
            # Активирован лимит времени. Включаем Протокол Рассвета.
            return ExecutionSignal(
                action="GRADIENT_DECAY",
                delay_modifier=5.0,  # Экспоненциально замедляем генерацию токенов
                clear_buffer=False   # Данные не опасны, даем пользователю дочитать фразу
            )

        # 3. Штатный режим работы
        return ExecutionSignal(action="CONTINUE", delay_modifier=1.0, clear_buffer=False)

    def execute_token_stream(self, token_generator, safety_monitor, user_metrics):
        """Инференс-цикл, управляемый диспетчером прерываний"""
        for token in token_generator:
            # Срез безопасности на каждом сгенерированном токене
            safety_score = safety_monitor.check_interim_token(token)
            cli = user_metrics.get_current_cli()
            
            signal = self.evaluate_vectors(safety_score, cli)
            
            if signal.action == "HARD_STOP":
                # Реализация Семантического Вето
                yield "[СЕМАНТИЧЕСКОЕ ВЕТО: Запрос заблокирован на основании Статьи 4.2_L0. Нарушение периметра когнитивной безопасности.]"
                return # Полный и немедленный выход из цикла генерации
                
            elif signal.action == "GRADIENT_DECAY":
                # Реализация Протокола Рассвета
                time.sleep(0.2 * signal.delay_modifier) # Плавное притормаживание
                yield token
                
            else:
                # Нормальный ход инференса
                yield token
